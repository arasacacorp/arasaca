   403→export default function Roadmap2Page() {
   404→  const STORAGE_KEY = "roadmap2-checked";
   405→
   406→  const [checked, setChecked] = useState<Set<string>>(new Set());
   407→  const [mounted, setMounted] = useState(false);
   408→
   409→  /* Load from localStorage */
   410→  useEffect(() => {
   411→    try {
   412→      const saved = localStorage.getItem(STORAGE_KEY);
   413→      if (saved) setChecked(new Set(JSON.parse(saved)));
   414→    } catch {}
   415→    setMounted(true);
   416→  }, []);
   417→
   418→  /* Save to localStorage */
   419→  useEffect(() => {
   420→    if (!mounted) return;
   421→    try {
   422→      localStorage.setItem(STORAGE_KEY, JSON.stringify([...checked]));
   423→    } catch {}
   424→  }, [checked, mounted]);
   425→
   426→  const toggle = (href: string) => {
   427→    setChecked((prev) => {
   428→      const next = new Set(prev);
   429→      if (next.has(href)) next.delete(href);
   430→      else next.add(href);
   431→      return next;
   432→    });
   433→  };
   434→
   435→  const resetAll = () => setChecked(new Set());
   436→
   437→  /* Totals */
   438→  const allHrefs: string[] = [];
   439→  categories.forEach((cat) => {
   440→    cat.links.forEach((l) => allHrefs.push(l.href));
   441→    cat.subs?.forEach((s) => s.links.forEach((l) => allHrefs.push(l.href)));
   442→  });
   443→  const totalLinks = allHrefs.length;
   444→  const checkedCount = allHrefs.filter((h) => checked.has(h)).length;
   445→  const progress = totalLinks > 0 ? Math.round((checkedCount / totalLinks) * 100) : 0;
   446→
   447→  return (
   448→    <main className="min-h-screen flex flex-col" style={{ background: C.muted }}>
   449→      {/* ══════════ HERO ══════════ */}
   450→      <section className="relative overflow-hidden pt-16 lg:pt-[120px] pb-14 md:pb-16" style={{ background: C.dark }}>
   451→        <div className="container-kept relative z-10 pt-6 md:pt-10 lg:pt-12">
   452→          <motion.nav
   453→            className="flex items-center gap-2 mb-6"
   454→            initial={{ opacity: 0 }}
   455→            animate={{ opacity: 1 }}
   456→            transition={{ duration: 0.5, delay: 0.1 }}
   457→          >
   458→            <Link href="/" className="text-[12px] transition-colors hover:text-white" style={{ color: "rgba(255,255,255,0.5)" }}>
   459→              Главная
   460→            </Link>
   461→            <ChevronRight className="w-3 h-3" style={{ color: "rgba(255,255,255,0.25)" }} />
