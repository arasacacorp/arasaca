    24→  client: string;
    25→  /** Год или период (опционально) */
    26→  year?: string;
    27→  /** Связанные услуги для блока внизу страницы кейса */
    28→  relatedServices?: CaseRelated[];
    29→}
    30→
    31→/** Структура первого кейса: Задача → Решение (список) → Результат */

    33→  structure: "task-solution-result";
    34→  task: string;
    35→  solution: string[];
    36→  result: string;
    37→}

    42→  sections: { title: string; content: string | string[] }[];
